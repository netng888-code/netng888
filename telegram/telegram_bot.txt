    """

    v1.4【重要bugfix】：舊版淨係睇「而家嘅ET日曆日期」係星期幾，完全

    預熱完全停擺，用戶會見到「見唔到任何alert」。

    修正邏輯（owning trading day）：
      - 週六全日                    → False（真正冇夜盤，
                                       Fri 20:00 ET收咗市直至Sun 20:00 ET）
      - 週日 00:00–19:59 ET          → False（仲係真正休市）
      - 週日 20:00–23:59 ET          → 呢段係「週一」嘅夜盤已經開始，
                                       改用「明日」（週一）嘅日期查假期表
      - 週五 20:00–23:59 ET          → False（冇夜盤，要等週日20:00
                                       先再開）
      - 週一至週五其餘時段           → 沿用原本邏輯，查當日假期表
    """
    et_now  = datetime.datetime.now(US_TZ)
    weekday = et_now.weekday()   # 0=週一 … 6=週日
    minutes = et_now.hour * 60 + et_now.minute

    if weekday == 5:              # 週六，全日冇夜盤
        return False

    if weekday == 6:              # 週日
        if minutes < 1200:        # 20:00 ET之前，仲係真休市
            return False
        # 20:00 ET之後 = 週一夜盤已開始，owning trading day係週一
        owning_day = et_now + datetime.timedelta(days=1)
        return owning_day.strftime("%Y-%m-%d") not in _NYSE_HOLIDAYS

    if weekday == 4 and minutes >= 1200:   # 週五20:00後，冇夜盤
        return False

    date_str = et_now.strftime("%Y-%m-%d")
    if date_str in _NYSE_HOLIDAYS:
        return False
    return True

# ET 交易時段判斷（v1.1新增）
# 分界點同 gex_chart_v4_terminal.html 嘅 detectSession()/getT1()
# 完全一致——兩邊如要調整分界，記得同步改，否則Bot同Dashboard
# 會喺唔同時段判斷出唔同session，數字對唔上。
# ═══════════════════════════════════════════════════
def detect_et_session() -> str:
    """
    返回：'weekend' | 'overnight' | 'premarket' | 'regular' | 'afterhours'

    v1.4【重要bugfix】：同 is_nyse_trading_day() 同一個根源bug——舊版
    淨係睇 et.weekday()>=5 就一律返回weekend，冇考慮夜盤交易實際上
    喺週日20:00 ET已經開始（屬於嚟緊嗰個週一），亦冇考慮週五20:00 ET
    收市後其實已經冇夜盤（下一節要等週日20:00先重開）。
    結果：舊版喺週日20:00–23:59 ET會錯誤返回"weekend"（應該係
    "overnight"），令 fetch_live_quote_futu/fetch_all_quotes_futu 揀
    錯欄位（跌returns返last_price而唔係overnight_price）；亦令週五
    20:00–23:59 ET錯誤返回"overnight"（應該係"weekend"，呢段根本
    冇夜盤數據）。

    分界（修正後）：
      週六全日                     → weekend
      週日 00:00–19:59 ET          → weekend
      週日 20:00–23:59 ET          → overnight（週一夜盤已開始）
      週五 20:00–23:59 ET          → weekend（冇夜盤，等週日20:00）
      週一至五 00:00–04:00 ET      → overnight
      週一至五 04:00–09:30 ET      → premarket
      週一至五 09:30–16:00 ET      → regular
      週一至五 16:00–20:00 ET      → afterhours
      週一至四 20:00–24:00 ET      → overnight（承接落去下一個交易日）
    用 zoneinfo 自動處理 EDT/EST，唔會夏令時轉錯。
    """
    et = datetime.datetime.now(US_TZ)
    wd = et.weekday()   # 0=週一 … 6=週日
    m  = et.hour * 60 + et.minute

    if wd == 5:                      # 週六，全日休市
        return "weekend"
    if wd == 6:                      # 週日
        return "overnight" if m >= 1200 else "weekend"
    if wd == 4 and m >= 1200:        # 週五20:00後，冇夜盤
        return "weekend"

    if m < 240:
        return "overnight"
    if m < 570:
        return "premarket"
    if m < 960:
        return "regular"
    if m < 1200:
        return "afterhours"
    return "overnight"


def _select_t1_price(rec: dict, sess: str) -> float | None:
    """
    根據時段揀返 Futu snapshot 入面啱嘅『即時』價格欄位。
    對應 gex_chart_v4_terminal.html 嘅 getT1()：
      premarket  → pre_price
      afterhours → after_price
      overnight  → overnight_price
      regular/weekend → last_price
    任何一個 extended-hours 欄位缺值，fallback返 last_price
    （好過整個報價冇晒）。
    """
    if sess == "premarket":
        return rec.get("pre_price") or rec.get("last_price")
    if sess == "afterhours":
        return rec.get("after_price") or rec.get("last_price")
    if sess == "overnight":
        return rec.get("overnight_price") or rec.get("last_price")
    return rec.get("last_price")

# Bot 層 GEX Live Cache（v1.12起 20小時TTL；v1.13起持久化到磁碟）
# 由 gex_refresh_job 每30分鐘嘗試預熱，alert job 直接讀取
# ═══════════════════════════════════════════════════
_gex_live_cache: dict[str, tuple[dict, float]] = {}  # sym → (gex_data, timestamp)
GEX_LIVE_CACHE_TTL = 72000  # 20 小時（v1.12起，原30分鐘，見上面註解）
GEX_LIVE_CACHE_FILE = "gex_live_cache.json"  # v1.13新增：磁碟持久化

def _load_gex_live_cache_from_disk():
    """
    v1.13新增：Bot啟動時讀返上次留低嘅GEX live cache，等process
    重啟唔會令20小時TTL形同虛設。讀取失敗（檔案未存在/格式壞咗）
    就當做空cache處理，唔會累到Bot完全啟動唔到。
    """
    global _gex_live_cache
    try:
        with open(GEX_LIVE_CACHE_FILE, "r", encoding="utf-8") as f:
            raw = json.load(f)
        _gex_live_cache = {sym: (v["data"], v["ts"]) for sym, v in raw.items()}
        print(f"   📦 已讀取磁碟GEX cache：{len(_gex_live_cache)}隻symbol")
    except Exception:
        _gex_live_cache = {}

def _save_gex_live_cache_to_disk():
    """寫入失敗（例如冇寫入權限）唔應該累到set_gex_live()本身失敗，
    靜默處理就算——下次成功refresh再試過寫，好過成個process炸咗。"""
    try:
        raw = {sym: {"data": data, "ts": ts} for sym, (data, ts) in _gex_live_cache.items()}
        with open(GEX_LIVE_CACHE_FILE, "w", encoding="utf-8") as f:
            json.dump(raw, f, ensure_ascii=False, indent=2)
    except Exception:
        pass

def get_gex_live(sym: str) -> dict | None:
    """讀取 bot 層 GEX cache，過期或無資料返回 None"""
    if sym not in _gex_live_cache:
        return None
    data, ts = _gex_live_cache[sym]
    if time.time() - ts > GEX_LIVE_CACHE_TTL:
        return None
    return data

def set_gex_live(sym: str, data: dict):
    _gex_live_cache[sym] = (data, time.time())
    _save_gex_live_cache_to_disk()  # v1.13新增：每次成功refresh都同步落磁碟

# .md 文件內存緩存（TTL=60分鐘）
# 背景警報每15分鐘 × 18隻 = 本來72次/小時，超出 GitHub 60次/小時免費限制
# 加緩存後：每隻每小時只讀 GitHub 一次（18次/小時），完全唔超限
_md_cache: dict[str, tuple[str, float]] = {}   # sym → (content, timestamp)
MD_CACHE_TTL = 3600  # 60 分鐘

def fetch_stock_md(sym: str, force_refresh: bool = False) -> str | None:
    now = time.time()  # v1.1: 頂部已有 import time，唔再需要局部 import 補救
    if not force_refresh and sym in _md_cache:
        content, ts = _md_cache[sym]
        if now - ts < MD_CACHE_TTL:
            return content

    url = (f"https://raw.githubusercontent.com/{GITHUB_OWNER}/"
           f"{GITHUB_REPO}/{GITHUB_BRANCH}/stocks/{sym}.md")
    try:
        r = requests.get(url, timeout=10)
        if r.status_code == 200:
            _md_cache[sym] = (r.text, now)
            return r.text
        return None
    except Exception:
        # 網絡失敗時用舊緩存（如果有），寧可返回過期數據都好過冇
        if sym in _md_cache:
            return _md_cache[sym][0]
        return None

def parse_cot_md(md: str) -> dict:
    """
    從 cftc_cot_updater.py 嘅 generate_cot_md() 生成嘅markdown抽返
    結構化數據，同parse_md()（持倉GEX快照用嗰個）同一套regex-based
    設計原則。

    抽取嘅欄位：
      report_date     (str)  數據截至嘅Report Date（週二倉位截止）
      long/short/net  (str)  Leveraged Funds Long/Short/Net（連
                              千分位逗號同"張"單位一齊keep住，直接
                              顯示用，唔做二次數值運算）
      net_pct_oi      (str)  Net佔總未平倉合約(OI)百分比
      open_interest   (str)  總未平倉合約
      wow             (str)  週對週(WoW)變化敘述（已經由
                              cftc_cot_updater.py嘅classify_wow_
                              change()分類好，呢度純粹讀取）
      cot_index_desc  (str)  COT Index分級敘述（同上，已經由
                              classify_cot_index()分類好）
      updated         (str)  最後更新時間戳
      recent_weeks    (list) 近8週mini table揀最尾4行，
                              [(date, net_str, wow_str), ...]
    """
    data = {}

    m = re.search(r'數據截至（Report Date，週二倉位截止）：\*\*([\d\-]+)\*\*', md)
    if m: data["report_date"] = m.group(1)

    m = re.search(r'Long（多倉） \| ([^\|\n]+) \|', md)
    if m: data["long"] = m.group(1).strip()

    m = re.search(r'Short（空倉） \| ([^\|\n]+) \|', md)
    if m: data["short"] = m.group(1).strip()

    m = re.search(r'\*\*Net（淨倉位）\*\* \| \*\*([^\|\n]+)\*\* \|', md)
    if m: data["net"] = m.group(1).strip()

    m = re.search(r'Net 佔總未平倉合約\(OI\) \| ([^\|\n]+) \|', md)
    if m: data["net_pct_oi"] = m.group(1).strip()

    m = re.search(r'總未平倉合約\(Open Interest\) \| ([^\|\n]+) \|', md)
    if m: data["open_interest"] = m.group(1).strip()

    m = re.search(r'週對週\(WoW\)變化 \| ([^\|\n]+) \|', md)
    if m: data["wow"] = m.group(1).strip()

    m = re.search(r'🎯 COT Index.*?\*\*(.+?)\*\*', md, re.S)
    if m: data["cot_index_desc"] = m.group(1).strip()

    m = re.search(r'最後更新：([\d\- :]+HKT)', md)
    if m: data["updated"] = m.group(1).strip()

    rows = re.findall(r'\| (\d{4}-\d{2}-\d{2}) \| ([^\|\n]+) \| ([^\|\n]+) \|', md)
    if rows:
        data["recent_weeks"] = rows[-4:]

    return data

# 報價：Futu Bridge（T1主要）→ Finnhub（fallback）
# ═══════════════════════════════════════════════════
def fetch_live_quote_futu(sym: str) -> dict | None:
    """
    主要報價來源：Futu Bridge /api/quote
    優點：T1真實即時價，盤前/正常/盤後/夜盤全時段皆係最新成交價。
    需要：futu_bridge.py + Futu OpenD 係運行狀態。

    v1.1修正：舊版一律讀 last_price，喺盤前/盤後/夜盤時段last_price
    其實仲停留喺上個regular session嘅收盤價（唔會即時更新），令
    「即時報價」名不副實。而家用 detect_et_session() 揀返啱嘅
    session-aware欄位（pre_price/after_price/overnight_price），
    同 gex_chart_v4_terminal.html 嘅 getT1() 邏輯一致。
    """
    try:
        r = requests.get(f"{FUTU_BRIDGE_URL}/api/quote", timeout=5)
        if r.status_code != 200:
            return None
        data    = r.json()
        us_list = data.get("us", [])
        target  = f"US.{sym.upper()}"
        sess    = detect_et_session()
        for rec in us_list:
            if rec.get("code") != target:
                continue
            price = _select_t1_price(rec, sess)
            prev  = rec.get("prev_close_price") or 0
            t2    = rec.get("last_price")  # 今日regular收盤價（唔隨session變）
            # 盤中/週末：同前日收盤(T3)比；盤前/盤後/夜盤：同今日收盤(T2)比
            # （同 gex_chart_v4_terminal.html 嘅 T1_chg_base 邏輯一致）
            base = prev if sess in ("regular", "weekend") else (t2 or prev)
            chg = pct = None
            if price and base:
                chg = round(float(price) - float(base), 4)
                pct = round(chg / float(base) * 100, 4) if float(base) != 0 else None
            if price and float(price) > 0:
                return {
                    "price":   float(price),
                    "change":  float(chg or 0),
                    "pct":     float(pct or 0),
                    "prev":    float(prev),
                    "session": sess,
                    "ts":      time.time(),
                    "source":  "futu_t1",
                }
        return None
    except Exception:
        return None


def fetch_live_quote_finnhub(sym: str) -> dict | None:
    """
    Fallback報價：Finnhub /quote（免費tier）。
    限制：盤前/盤後可能仍返前收市價，timestamp需自行判斷新舊。
    """
    url    = "https://finnhub.io/api/v1/quote"
    params = {"symbol": sym, "token": FINNHUB_KEY}
    try:
        r = requests.get(url, params=params, timeout=10)
        if r.status_code == 200:
            d = r.json()
            return {
                "price":  d.get("c", 0),
                "change": d.get("d", 0),
                "pct":    d.get("dp", 0),
                "prev":   d.get("pc", 0),
                "ts":     d.get("t", 0),
                "source": "finnhub",
            }
        return None
    except Exception:
        return None



def fetch_all_quotes_futu() -> dict[str, dict]:
    """
    一個 API call 攞晒 US_HOLDINGS 全部即時報價（Futu T1）。
    比逐隻 call 高效得多，背景 job 同 /report 首選。
    返回 {sym: {price, change, pct, prev, session, source}} 或 {} (Futu離線)

    v1.1修正：同 fetch_live_quote_futu 一樣，唔再一律讀last_price，
    改用 detect_et_session() + _select_t1_price() 揀返啱嘅欄位。
    """
    try:
        r = requests.get(f"{FUTU_BRIDGE_URL}/api/quote", timeout=8)
        if r.status_code != 200:
            return {}
        us_list = r.json().get("us", [])
        sess    = detect_et_session()
        result  = {}
        for rec in us_list:
            code = rec.get("code", "")
            if not code.startswith("US."):
                continue
            sym   = code[3:]
            price = _select_t1_price(rec, sess)
            prev  = rec.get("prev_close_price") or 0
            t2    = rec.get("last_price")
            base  = prev if sess in ("regular", "weekend") else (t2 or prev)
            chg = pct = None
            if price and base:
                chg = round(float(price) - float(base), 4)
                pct = round(chg / float(base) * 100, 4) if float(base) != 0 else None
            if price and float(price) > 0:
                result[sym] = {
                    "price":   float(price),
                    "change":  float(chg or 0),
                    "pct":     float(pct or 0),
                    "prev":    float(prev),
                    "session": sess,
                    "ts":      time.time(),
                    "source":  "futu_t1",
                }
        return result
    except Exception:
        return {}


def fetch_live_quote(sym: str) -> dict | None:
    """
    統一報價入口：Futu T1（主）→ Finnhub（fallback）。
    呼叫方唔需關心底層來源，直接用返回嘅 price/change/pct/source。
    """
    result = fetch_live_quote_futu(sym)
    if result:
        print(f"   [Quote] {sym} <- Futu T1  ${result['price']}")
        return result
    result = fetch_live_quote_finnhub(sym)
    if result:
        print(f"   [Quote] {sym} <- Finnhub(fallback)  ${result['price']}")
        return result
    return None

def market_session_label(quote_ts: int = 0, source: str = "", session: str | None = None) -> str:

    修正：加返休市判斷，一律行is_nyse_trading_day()（已經包埋
    _NYSE_HOLIDAYS假期list + 週末，比detect_et_session()淨睇時分

      - 唔係交易日（假期或週末）→ 唔理報價年齡，一律顯示友善提示

      - 係交易日（包括夜盤/盤前/盤後/盤中）但報價仍然過舊 → 先顯示

    """

    # ── 休市時段（假期或週末）：唔理報價年齡，一律用友善提示 ──
    # is_nyse_trading_day()已經包埋_NYSE_HOLIDAYS，比呢度sess=='weekend'
    # 呢個純時分判斷更全面（平日假期detect_et_session()驗唔到）。
    if not is_nyse_trading_day() or sess == "weekend":
        et_now       = datetime.datetime.now(US_TZ)
        closed_label = "週末休市" if et_now.weekday() >= 5 else "假期休市"
        source_tag   = " [Futu T1]" if source == "futu_t1" else (" [Finnhub]" if source == "finnhub" else "")
        return f"📅 {closed_label}，顯示上一交易日收市數據{source_tag}"

    # ── 交易日內：Futu T1直接信任；Finnhub fallback先做新舊判斷 ──
    if source != "futu_t1" and quote_ts:
        now = datetime.datetime.now()
        quote_dt = datetime.datetime.fromtimestamp(quote_ts)
        age_min  = (now - quote_dt).total_seconds() / 60
        if age_min > 15:
            age_str = f"{int(age_min)}分鐘前" if age_min < 60 else f"{age_min/60:.1f}小時前"
            return f"⚠️ 非即時報價（{age_str}，可能係收市價）[Finnhub]"

    label      = _SESSION_LABELS.get(sess, sess)
    source_tag = " [Futu T1]" if source == "futu_t1" else (" [Finnhub]" if source == "finnhub" else "")
    return f"{label}{source_tag}"

# 解析 .md 內容，抽出關鍵數字（供 /pnl /report 計算用）
# ═══════════════════════════════════════════════════
def parse_md(md: str) -> dict:
    """從 generate_md() 生成嘅 markdown 抽返結構化數據"""
    data = {}

    m = re.search(r'現價 \| \*\*\$([\d,]+\.?\d*)\*\*', md)
    if m: data["price"] = float(m.group(1).replace(",", ""))

    m = re.search(r'未實現盈虧 \| (▲|▼) \$([\d,]+)', md)
    if m:
        data["pnl_sign"] = 1 if m.group(1) == "▲" else -1
        data["pnl"] = float(m.group(2).replace(",", ""))

    m = re.search(r'Gamma Flip \| \$([\d,]+\.?\d*)', md)
    if m: data["flip"] = float(m.group(1).replace(",", ""))

    m = re.search(r'Put Wall（支撐） \| \$([\d,]+\.?\d*)', md)
    if m: data["put_wall"] = float(m.group(1).replace(",", ""))

    m = re.search(r'Call Wall（阻力） \| \$([\d,]+\.?\d*)', md)
    if m: data["call_wall"] = float(m.group(1).replace(",", ""))

    m = re.search(r'狀態[：:]\s*\*{0,2}(.+?)\*{0,2}\s*$', md, re.MULTILINE)
    if m: data["status"] = m.group(1).strip()

    m = re.search(r'最後更新：([\d\- :]+HKT)', md)
    if m: data["updated"] = m.group(1).strip()

    return data

# 發送AI生成文字（v1.15新增）── Markdown parse失敗自動fallback
# ═══════════════════════════════════════════════════
async def safe_reply_markdown(message_obj, text: str):
    """
    用 parse_mode="Markdown" 發送AI生成嘅自由文字（/gex同/ask嘅AI分析
    步驟都用緊）。

    背景【重要bugfix】：用戶截圖+終端log證實/ask AVGO／/ask SPCX
    卡死喺「分析中，請稍候...」永遠冇下文，終端log顯示：
      telegram.error.BadRequest: Can't parse entities: can't find
      end of the entity starting at byte offset 858
    根源：DeepSeek/Kimi生成嘅自由文字入面，如果剛好出現冇配對好嘅
    Markdown特殊字符（單獨一個*、_、`、[ 之類，例如AI寫「*」表示
    項目符號但漏咗閉合嗰個，或者股票名/數字入面剛好夾埋一個特殊
    符號），Telegram官方Markdown parser會直接拒絕成個message、
    完全唔會送出——同message內容有幾長、AI分析講咗幾多嘢完全無關，
    純粹嗰一個字符配對唔到就死。cmd_gex()同cmd_ask()嘅AI分析步驟
    舊版都直接
      `await update.message.reply_text(f"...{analysis}...",
                                        parse_mode="Markdown")`
    冇任何try/except包住——一旦AI呢次生成嘅文字剛好觸發呢個parse
    錯誤，個await直接拋BadRequest，成個command coroutine喺嗰度
    死咗（PTB嘅Application冇註冊error handler，淨係log低exception，
    唔會累到成個bot process崩潰，但呢條command嘅使用者見到嘅就係
    「分析中...」之後永遠冇下文）。LLM生成嘅自由文字冇任何辦法
    保證每次輸出嘅Markdown符號都啱啱好配對晒，呢類bug喺embed AI
    output落Telegram Markdown message嘅位一定會不定期撞到，唔係
    一次性修正就永遠唔會再發生，所以要做返呢一層防禦。

    修正：捕捉BadRequest，如果錯誤訊息確認係entity parse失敗，
    改用冇parse_mode（純文字）重新發送一次——寧願AI分析嗰段文字
    冇晒bold/斜體格式，都好過成個回覆完全送唔到用戶手上見唔到
    任何嘢。非parse相關嘅其他BadRequest（例如message太長）維持
    原本咁raise，唔會被呢層fallback静默吞咗。
    """
    try:
        await message_obj.reply_text(text, parse_mode="Markdown")
    except BadRequest as e:
        if "can't parse entities" in str(e).lower():
            print(f"   ⚠️ Markdown parse失敗，改用純文字重發: {e}")
            await message_obj.reply_text(text)
        else:
            raise

# OpenRouter → DeepSeek V4 Flash 0731（主）／Kimi K2.5（fallback）：生成分析建議
# ═══════════════════════════════════════════════════
def ask_ai(prompt: str, model: str | None = None, _is_fallback_retry: bool = False) -> str:
    """
    v1.2新增：model參數 + 自動fallback。
    若 ANALYSIS_MODEL（主，v1.6起係DeepSeek V4 Flash）被OpenRouter
    帳戶所屬地區封鎖（403）或連接逾時，自動retry一次改用
    FALLBACK_MODEL（v1.6起係Kimi K2 Thinking），只做一次retry
    （_is_fallback_retry防止死循環），避免主model有事就成個AI
    分析功能靜默停擺。
    401（key本身無效）／402（餘額不足）／429（rate limit）呢三類
    錯誤屬於帳戶層面，換model都冇用，唔會觸發fallback。

    v1.8【重要bugfix】：實測/gex查詢AI分析步驟，Telegram顯示成句
    「🤖 AI分析：\nNone」——即係choices[0]["message"]["content"]
    本身係Python None，唔係空字串，先會令f-string串出字面「None」
    （之前呢個function冇檢查content係咪None/空，成功狀態碼200就
    直接return content，冇診斷過）。
    根源：ANALYSIS_MODEL喺v1.8升做deepseek-v4-flash-0731——呢個
    係reasoning model（OpenRouter model卡明確標示支援high/xhigh
    reasoning effort），reasoning階段本身會食用max_tokens呢個共用
    budget。舊版max_tokens=500對reasoning model嚟講太細，reasoning
    階段直接食晒成個budget，令final answer嗰邊response嘅
    message.content變成JSON null（Python讀出嚟就係None）——呢個
    唔係DeepSeek獨有，OpenRouter官方reasoning tokens文件都有講
    低budget會撞到呢個症狀（"the reasoning phase can consume the
    entire max_tokens budget, leaving zero tokens for the final
    answer"）。futu_bridge_v2.py嘅AI分析卡（GEX+K線terminal個
    網頁版）同一日撞到一模一樣嘅bug，修法完全一致，兩邊同步。
    修正：
      (1) max_tokens 500→2000——加大budget，成本可忽略（DeepSeek
          V4 Flash 0731約$0.09/$0.18 per M tokens）
      (2) 加返OpenRouter官方統一reasoning參數 reasoning:{"effort":
          "low","exclude":true}，盡量壓低reasoning消耗（model唔
          支援"low"嘅話OpenRouter自動map去佢支援嘅級別，唔會報錯）
      (3) content經None/空字串檢查後，當做同403/逾時同一類「呢個
          model今次搞唔掂」情況，自動fallback去FALLBACK_MODEL——
          唔會再淨係將None/空白直接return俾用戶見到。
    """
    url = "https://openrouter.ai/api/v1/chat/completions"
    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY}",
        "Content-Type": "application/json",
        "HTTP-Referer": "https://github.com/netng888-code/netng888",
        "X-Title": "GEX Monitor Bot",
    }
    use_model = model or ANALYSIS_MODEL
    payload = {
        "model": use_model,
        "max_tokens": 2000,
        "reasoning": {"effort": "low", "exclude": True},
        "messages": [{"role": "user", "content": prompt}],
    }
    try:
        r = requests.post(url, headers=headers, json=payload, timeout=30)

        # 一律先印完整response到終端，方便排查任何狀態碼
        print(f"[OpenRouter] model={use_model}  status={r.status_code}  body={r.text[:500]}")

        if r.status_code == 200:
            result = r.json()
            choices = result.get("choices", [])
            if not choices:
                return f"⚠️ AI回傳空結果，完整回應：{str(result)[:200]}"
            content = (choices[0].get("message", {}).get("content") or "").strip()
            if content:
                return content
            # content係None或空字串——大機會reasoning token食晒budget，
            # 當做失敗處理，同403/逾時一樣觸發自動fallback
            print(f"   ⚠️ {use_model} 回傳content為空/None（reasoning token可能耗盡預算）")
            if not _is_fallback_retry and use_model != FALLBACK_MODEL:
                print(f"   ⚠️ 自動fallback去 {FALLBACK_MODEL}")
                return ask_ai(prompt, model=FALLBACK_MODEL, _is_fallback_retry=True)
            return "⚠️ AI回傳content為空（reasoning token可能耗盡預算，兩個model都試過）"
        else:
            error_detail = ""
            try:
                error_json = r.json()
                error_detail = error_json.get("error", {}).get("message", r.text[:300])
            except Exception:
                error_detail = r.text[:300]

            # 403 = 呢隻model/provider喺你個OpenRouter帳戶地區被封鎖
            # ——自動fallback去FALLBACK_MODEL，唔使等你手動改設定先發現
            if r.status_code == 403 and not _is_fallback_retry and use_model != FALLBACK_MODEL:
                print(f"   ⚠️ {use_model} 被拒絕[403]，自動fallback去 {FALLBACK_MODEL}")
                return ask_ai(prompt, model=FALLBACK_MODEL, _is_fallback_retry=True)

            if r.status_code == 401:
                return "⚠️ AI API認證失敗：OPENROUTER_API_KEY無效或未填入"
            elif r.status_code == 402:
                return "⚠️ OpenRouter帳戶餘額不足，請去openrouter.ai充值"
            elif r.status_code == 403:
                return f"⚠️ AI API被拒絕 [403]（{use_model}，fallback亦失敗）：{error_detail}"
            elif r.status_code == 429:
                return "⚠️ 請求過於頻密，請稍後再試"
            else:
                return f"⚠️ AI API錯誤 [{r.status_code}]（{use_model}）：{error_detail}"
    except requests.exceptions.Timeout:
        if not _is_fallback_retry and use_model != FALLBACK_MODEL:
            print(f"   ⚠️ {use_model} 連接逾時，自動fallback去 {FALLBACK_MODEL}")
            return ask_ai(prompt, model=FALLBACK_MODEL, _is_fallback_retry=True)
        return "⚠️ AI API連接逾時（30秒），請稍後再試"
    except Exception as e:
        return f"⚠️ AI API連接失敗：{type(e).__name__}: {e}"

def compute_wall_status(price: float | None, flip: float | None,
                         put_wall: float | None, call_wall: float | None) -> str:
    """
    v1.2新增，AVGO誤判事件嘅核心修正。

    背景：build_gex_prompt() 舊版淨係將現價/Put Wall/Call Wall三個
    原始數字寫落prompt文字，叫AI自己計百分比、自己判斷邊個高邊個
    低。實測AVGO現價$383.01 vs Put Wall $385.00，AI計啱咗百分比
    （0.5%）但方向判斷反咗——講成「接近支撐位」，但其實Put Wall
    ($385) > 現價($383.01)，即係支撐已經跌穿，唔係「接近」。

    v1.13：cmd_ask()（/ask指令）一直冇call呢個function——用戶截圖
    證實/ask AVGO分析出現「股價392.75低於Put Wall(410)/Call
    Wall(420)嘅區間，卻話「已逼近Call牆」、仲話係「正Gamma區」」，
    兩者都同Put Wall/Gamma Flip嘅實際數值關係矛盾，AI自己計方向
    計錯咗，同呢度嘅AVGO事件屬同一種錯誤，只不過發生喺/ask呢條
    冇受呢個function保護嘅code path。而家cmd_ask()已經改用返呢個
    function（見下面改動）。

    修正原則：方向判斷（已跌穿/未跌穿/邊個高邊個低）呢類邏輯運算
    一定要喺Python code度做，用普通if/else保證100%準確；唔可以將
    呢類判斷交低LLM喺自然語言入面自己推斷——呢個先係徹底根治嘅
    方法，同用邊隻AI model冇關係（Kimi/DeepSeek/其他model一樣可能
    喺呢類任務中犯呢種方向性錯誤，分別只在於機率高低）。

    返回一段冇歧義嘅中文敘述，直接交畀AI，AI唔使再自己計算方向，
    淨係負責將已確定嘅事實組織做操作建議。
    """
    if not price or price <= 0:
        return "現價數據暫缺，無法判斷相對GEX水平位置。"

    parts = []

    if put_wall:
        diff_put = price - put_wall
        pct_put  = abs(diff_put) / price * 100
        if diff_put < 0:
            parts.append(
                f"現價已跌穿Put Wall支撐（${put_wall:,.2f}），低咗{pct_put:.2f}%"
                f"——支撐位已失守，唔係「接近支撐」"
            )
        elif pct_put < 3:
            parts.append(
                f"現價企穩喺Put Wall（${put_wall:,.2f}）之上，僅高{pct_put:.2f}%，"
                f"非常接近支撐位但未跌穿"
            )
        else:
            parts.append(f"現價高於Put Wall（${put_wall:,.2f}）{pct_put:.2f}%，未逼近支撐")

    if call_wall:
        diff_call = call_wall - price
        pct_call  = abs(diff_call) / price * 100
        if diff_call < 0:
            parts.append(
                f"現價已突破Call Wall阻力（${call_wall:,.2f}），高咗{pct_call:.2f}%"
                f"——阻力位已被攻破"
            )
        elif pct_call < 3:
            parts.append(f"現價逼近Call Wall（${call_wall:,.2f}），僅差{pct_call:.2f}%，接近關鍵阻力")
        else:
            parts.append(f"現價低於Call Wall（${call_wall:,.2f}）{pct_call:.2f}%，未逼近阻力")

    if flip:
        if price >= flip:
            parts.append(f"現價處於Gamma Flip（${flip:,.2f}）之上，屬正Gamma區（波動相對受抑制）")
        else:
            parts.append(f"現價處於Gamma Flip（${flip:,.2f}）之下，屬負Gamma區（波動可能放大）")

    return "；".join(parts) + "。" if parts else "GEX水平數據不足，無法判斷位置。"


def compute_breakout_scenario(dealer_bias: str) -> str:
    """
    v1.16新增，修正一個現存bug：build_gex_prompt()嘅第2、3點（舊版）
    無條件寫死「突破Call Wall→Gamma Squeeze加速升勢」、「跌穿Put
    Wall→拋售鏈加速跌勢」呢個敘事，完全冇理會現時究竟係Long Gamma
    定Short Gamma——但呢個codebase自己嘅慣例（compute_wall_status()
    「正Gamma區＝波動相對受抑制」、dealer_bias標籤「Long Gamma＝
    市場穩定」）已經講明：Long Gamma環境下Dealer避險方向應該係
    「壓抑」（漲就沽、跌就買，mean-reverting），唔係「加速」——舊版
    prompt對Long/Short Gamma兩種情況一律用同一套「加速」敘事，喺
    Long Gamma嗰種情況底下方向係講反咗（實測案例：PLTR Net GEX
    +100.7M／Long Gamma，AI舊版分析依然講「升穿Call Wall會Gamma
    Squeeze加速噴上」、「跌穿Put Wall會拋售鏈加速下跌」，同「Long
    Gamma＝市場穩定」呢個系統自己嘅結論互相矛盾）。

    呢個function同compute_wall_status()一樣嘅設計原則：方向性敘事
    一律由Python code根據dealer_bias預先寫定，AI唔准自己另外推斷
    方向，只可以引用同做語言潤飾。

    Long Gamma（dealer整體傾向壓抑波動）：
      - 突破Call Wall：Dealer對沖行為（漲沽跌買）本身傾向壓抑，
        突破更多反映真實買盤力度蓋過壓抑力，而唔係人為加速；個別
        重倉strike嘅局部對沖流可能帶嚟短暫波動，但唔等於全面加速
      - 跌穿Put Wall：同樣道理，喺仍然處於Long Gamma嘅範圍內，
        Dealer對沖（跌買）傾向緩衝跌勢，真正嘅「加速下跌」通常要
        等價格進一步跌穿Gamma Flip先會出現（可對照上面wall_status
        入面Gamma Flip嘅實際位置嚟判斷仲有幾遠）

    Short Gamma（dealer整體傾向加速波動）：
      - 突破Call Wall：Dealer為對沖上升delta被迫追貨，形成典型
        Gamma Squeeze，升勢確實有機會加速
      - 跌穿Put Wall：Dealer為對沖被迫沽貨，形成拋售鏈，跌勢確實
        有機會加速

    dealer_bias為空／未知：保留中性、雙向都可能嘅籠統講法，唔強行
    斷定方向。
    """
    if dealer_bias == "LONG_GAMMA":
        return (
            "目前屬Long Gamma（正Gamma）狀態——Dealer對沖行為理論上傾向"
            "「壓抑」波動（股價升，Dealer沽貨；股價跌，Dealer買貨），"
            "呢個同市場穩定嘅描述一致。突破Call Wall更大機會反映真實"
            "買盤力度蓋過Dealer壓抑性沽壓，而唔係人為加速噴升；跌穿"
            "Put Wall喺仍然身處Long Gamma範圍內，Dealer買貨對沖亦傾向"
            "緩衝跌勢，真正嘅加速下跌通常要價格進一步跌穿Gamma Flip"
            "先會出現——分析呢兩個情境時必須以「壓抑」為預設方向，"
            "唔可以講成「突破/跌穿即加速」。"
        )
    if dealer_bias == "SHORT_GAMMA":
        return (
            "目前屬Short Gamma（負Gamma）狀態——Dealer對沖行為理論上"
            "傾向「加速」波動（股價升，Dealer被迫追貨；股價跌，Dealer"
            "被迫沽貨），呢個同波動放大嘅描述一致。突破Call Wall有機會"
            "觸發Dealer追貨形成Gamma Squeeze，跌穿Put Wall有機會觸發"
            "Dealer沽貨形成拋售鏈，兩個方向都可以用「加速」嚟描述。"
        )
    return (
        "Dealer Gamma狀態暫缺，突破/跌穿之後Dealer對沖行為究竟係"
        "壓抑定加速波動未能確定，分析時唔好單方面斷定「必然加速」，"
        "可以雙向討論但要註明呢個不確定性。"
    )


def build_gex_prompt(sym: str, name: str, data: dict, live: dict | None, session: str, source: str = "") -> str:

# GEX 橫向條形圖（matplotlib，暗色主題）
# ═══════════════════════════════════════════════════
def generate_gex_chart(sym: str, gex_data: dict, live_price: float | None = None) -> bytes | None:
    """
    從 Futu bridge /api/gex/{sym} 的 gex_by_strike 生成 PNG 圖表。
    返回 PNG bytes 或 None（matplotlib 未安裝 / 數據不足時）。
    """
    try:
        import matplotlib
        matplotlib.use("Agg")
        import matplotlib.pyplot as plt
        import matplotlib.patches as mpatches
        from io import BytesIO
    except ImportError:
        return None

    strikes_data = gex_data.get("gex_by_strike", [])
    if not strikes_data:
        return None

    spot = float(gex_data.get("spot") or live_price or 0)

    # 只顯示 spot ±30% 範圍（過濾遠端 OTM 雜訊）
    if spot > 0:
        strikes_data = [s for s in strikes_data
                        if spot * 0.70 <= s["strike"] <= spot * 1.30]

    # 最多40個strike，太多圖太密
    if len(strikes_data) > 40:
        strikes_data = sorted(strikes_data, key=lambda x: abs(x["strike"] - spot))[:40]
        strikes_data = sorted(strikes_data, key=lambda x: x["strike"])

    if not strikes_data:
        return None

    strikes  = [s["strike"]  for s in strikes_data]
    gex_vals = [s["net_gex"] for s in strikes_data]
    colors   = ["#00d4aa" if v >= 0 else "#ff4d6d" for v in gex_vals]

    fig_h = max(6, min(len(strikes) * 0.28, 14))
    fig, ax = plt.subplots(figsize=(9, fig_h))
    fig.patch.set_facecolor("#0a0c10")
    ax.set_facecolor("#0f1218")

    y_pos = list(range(len(strikes)))
    ax.barh(y_pos, gex_vals, color=colors, alpha=0.85, height=0.72)
    ax.set_yticks(y_pos)
    ax.set_yticklabels([f"${s:,.0f}" for s in strikes], fontsize=7, color="#c8d4e8")

    # 關鍵水平線（根據 strike 索引插值）
    def price_to_y(price):
        if price is None or not strikes:
            return None
        diffs = [abs(s - price) for s in strikes]
        return diffs.index(min(diffs))

    flip      = gex_data.get("flip")
    put_wall  = gex_data.get("put_wall")
    call_wall = gex_data.get("call_wall")
    current   = live_price or spot

    # v1.7：現價橫線顏色由 #f5a623（琥珀色）改做 #ffffff（白色）——
    # 用戶反映同下面Gamma Flip線嘅 #ffd700（金黃色）視覺上太接近，
    # 細圖睇唔清邊條係邊條（見附圖SKHY GEX chart，現價線同Flip線
    # 幾乎撞色）。白色對比度更高，一眼分辨得到；legend_patches用
    # 返呢個color變量生成圖例swatch，唔使額外改。
    line_defs = [
        (current,   "#ffffff", f"現價 ${current:,.1f}",           "--", 2.0),
        (flip,      "#ffd700", f"Gamma Flip ${flip:,.1f}" if flip else None,       "-.", 1.5),
        (put_wall,  "#ff4d6d", f"Put Wall ${put_wall:,.1f}" if put_wall else None,  ":",  1.5),
        (call_wall, "#00d4aa", f"Call Wall ${call_wall:,.1f}" if call_wall else None, ":", 1.5),
    ]
    legend_patches = []
    for price, color, label, ls, lw in line_defs:
        if price is not None:
            yi = price_to_y(price)
            if yi is not None:
                ax.axhline(y=yi, color=color, linestyle=ls, linewidth=lw, alpha=0.9)
            if label:
                legend_patches.append(mpatches.Patch(color=color, label=label))

    ax.axvline(x=0, color="#4a5568", linewidth=0.8)
    ax.set_xlabel("Net GEX (百萬 $)", color="#c8d4e8", fontsize=9)
    ax.tick_params(colors="#c8d4e8", labelsize=8)
    for sp in ax.spines.values():
        sp.set_edgecolor("#1e2430")

    bias     = gex_data.get("dealer_bias", "")
    bias_lbl = "📈 Long Gamma" if bias == "LONG_GAMMA" else "📉 Short Gamma"
    net_v    = gex_data.get("net_gex") or 0
    ax.set_title(
        f"{sym}  GEX by Strike     {bias_lbl}     Net GEX: {net_v:+.1f}M",
        color="white", fontsize=10, fontweight="bold", pad=10
    )

    if legend_patches:
        ax.legend(handles=legend_patches, loc="lower right", fontsize=7,
                  facecolor="#0f1218", labelcolor="#c8d4e8", edgecolor="#1e2430")

    plt.tight_layout()
    buf = BytesIO()
    plt.savefig(buf, format="png", dpi=110, bbox_inches="tight",
                facecolor=fig.get_facecolor())
    plt.close(fig)
    buf.seek(0)
    return buf.read()

KLINE_PERIOD_ALIASES = {
    "1": "1", "5": "5", "15": "15", "30": "30", "60": "60",
    "D": "D", "1D": "D", "DAY": "D",
    "W": "W", "1W": "W", "WEEK": "W",
    "M": "M", "1M": "M", "MONTH": "M",
}
KLINE_INTRADAY_PERIODS = {"1", "5", "15", "30", "60"}


def fetch_futu_kline(sym: str, period: str = "D", count: int = 90) -> dict | None:
    """
    呼叫本機 Futu Bridge /api/kline/{sym}（見futu_bridge_v2.py，已有
    MA9/20/60/布林線/RSI，唔使改bridge），攞K線數據俾/gex畫candlestick
    圖用。period同bridge一致：1/5/15/30/60（分鐘）／D／W／M。
    返回完整dict（bars/ma9/ma20/ma60/bb/rsi）或None（bridge離線/
    symbol錯/K線為空）。

    v1.15：加返print診斷——舊版任何失敗（連唔到bridge/K線為空/
    非200狀態碼）一律靜默return None，用戶完全睇唔到K線圖，亦冇
    任何終端log提示係邊一步失敗，好難診斷。而家統一將失敗原因
    print出嚟。
    """
    try:
        r = requests.get(
            f"{FUTU_BRIDGE_URL}/api/kline/{sym.upper()}",
            params={"period": period, "count": count},
            timeout=20,
        )
        if r.status_code == 200:
            data = r.json()
            if data.get("bars"):
                return data
            print(f"   ⚠️ fetch_futu_kline({sym}) 200 OK但bars為空（{data.get('error', '無error欄位')}）")
            return None
        print(f"   ⚠️ fetch_futu_kline({sym}) HTTP {r.status_code}")
        return None
    except Exception as e:
        print(f"   ⚠️ fetch_futu_kline({sym}) 連接失敗: {type(e).__name__}: {e}")
        return None


def generate_kline_chart(sym: str, kline_data: dict, gex_data: dict | None = None) -> bytes | None:
    """
    用 mplfinance 畫暗色主題candlestick K線圖，配色風格對齊
    generate_gex_chart()（漲#00d4aa／跌#ff4d6d／背景#0a0c10）：
      - 疊加 MA9(琥珀#f5a623) / MA20(藍#4a9eff) / MA60(紫#a78bfa)
      - 底部帶成交量subplot（mplfinance volume=True）
      - 若有gex_data（futu即時GEX），將Gamma Flip(白)/Call Wall(綠)/
        Put Wall(紅)/Max Pain(粉紅)四條水平線直接疊落K線圖——同
        gex_chart_v5_terminal.html嘅K線+GEX橫線做法一致，等用戶一張
        圖睇晒現價相對Wall嘅位置點樣隨時間演變。
    返回PNG bytes，或None（mplfinance未安裝／bars為空）。

    v1.15【重要bugfix】：用戶反映升級到v1.14後/gex完全冇K線圖，但
    終端log冇任何相關error——根源就係呢度：mplfinance係v1.14先加
    嘅新pip依賴，如果本機冇裝，`import mplfinance`拋ImportError，
    舊版export except ImportError: return None，完全靜默、冇print
    任何嘢，令人以為個功能「冇做嘢」而唔知道係缺咗依賴。而家加返
    一句print，第一次撞到就會喺終端見到清楚提示。
    """
    try:
        import mplfinance as mpf
        import pandas as pd
        import matplotlib.pyplot as plt
        from io import BytesIO
    except ImportError as e:
        print(f"   ⚠️ generate_kline_chart({sym}) 缺依賴: {e}（請 pip install mplfinance pandas）")
        return None

    bars = kline_data.get("bars", [])
    if not bars:
        return None

    period = kline_data.get("period", "D")
    df = pd.DataFrame(bars)
    if period in KLINE_INTRADAY_PERIODS:
        df["Date"] = pd.to_datetime(df["time"], unit="s")
    else:
        df["Date"] = pd.to_datetime(df["time"])
    df = df.set_index("Date")
    df = df.rename(columns={"open": "Open", "high": "High", "low": "Low",
                             "close": "Close", "volume": "Volume"})
    df = df[["Open", "High", "Low", "Close", "Volume"]]

    add_plots = []
    for key, color in (("ma9", "#f5a623"), ("ma20", "#4a9eff"), ("ma60", "#a78bfa")):
        vals = kline_data.get(key)
        if vals and len(vals) == len(df):
            s = pd.Series(vals, index=df.index, dtype="float64")
            if s.notna().any():
                add_plots.append(mpf.make_addplot(s, color=color, width=1.1))

    hline_prices, hline_colors = [], []
    if gex_data:
        for key, color in (("flip", "#ffffff"), ("call_wall", "#00d4aa"),
                            ("put_wall", "#ff4d6d"), ("max_pain", "#ec4899")):
            v = gex_data.get(key)
            if v:
                hline_prices.append(v)
                hline_colors.append(color)

    mc = mpf.make_marketcolors(up="#00d4aa", down="#ff4d6d", edge="inherit",
                                wick="inherit", volume="inherit")
    style = mpf.make_mpf_style(
        marketcolors=mc, facecolor="#0f1218", edgecolor="#1e2430",
        figcolor="#0a0c10", gridcolor="#141820", gridstyle="-",
        rc={"axes.labelcolor": "#c8d4e8", "xtick.color": "#4a5568",
            "ytick.color": "#4a5568", "text.color": "#c8d4e8"},
    )

    plot_kwargs = dict(
        type="candle", style=style, volume=True, figsize=(10, 7.5),
        title=f"\n{sym}  K線（{period}）",
        returnfig=True,
    )
    if add_plots:
        plot_kwargs["addplot"] = add_plots
    if hline_prices:
        plot_kwargs["hlines"] = dict(hlines=hline_prices, colors=hline_colors,
                                      linestyle="--", linewidths=1.3)

    try:
        fig, axes = mpf.plot(df, **plot_kwargs)
    except Exception as e:
        print(f"   ⚠️ generate_kline_chart({sym}) 繪圖失敗: {type(e).__name__}: {e}")
        return None

    buf = BytesIO()
    fig.savefig(buf, format="png", dpi=110, bbox_inches="tight",
                facecolor=fig.get_facecolor())
    plt.close(fig)
    buf.seek(0)
    return buf.read()

# /report 指令：今日全部17隻摘要
# ═══════════════════════════════════════════════════
async def cmd_report(update: Update, context: ContextTypes.DEFAULT_TYPE):
    register_chat(update.effective_chat.id)
    await update.message.reply_text("⏳ 生成今日報告（17隻持倉）...")

    # ── 一個call攞晒17隻即時報價（Futu T1，最高效）──
    quotes = fetch_all_quotes_futu()
    futu_ok = bool(quotes)

    # Futu離線時 fallback Finnhub（逐隻call，慢但穩）
    def get_price(sym: str) -> dict | None:
        if sym in quotes:
            return quotes[sym]
        return fetch_live_quote(sym)

    lines     = [f"📋 *今日持倉GEX報告* — {datetime.date.today()}"]
    src_label = "Futu T1" if futu_ok else "Finnhub"
    lines.append(f"_報價來源：{src_label}_\n")

    total_pnl   = 0
    total_cost  = 0
    total_value = 0

    for h in ALL_HOLDINGS:
        sym  = h["sym"]
        name = h["name"]
        qty  = h["qty"]
        cost = h["cost"]

        # 1. 即時報價
        live  = get_price(sym)
        price = live["price"] if live and live.get("price") else None

        # 2. GEX牆：Futu Bridge 即時（5分鐘cache in bridge）
        gex = fetch_futu_gex(sym)
        if gex and gex.get("call_wall"):
            call_wall = gex.get("call_wall")
            put_wall  = gex.get("put_wall")
            flip      = gex.get("flip")
            bias      = gex.get("dealer_bias", "")
        else:
            # Fallback：讀.md緩存
            md = fetch_stock_md(sym)
            md_data   = parse_md(md) if md else {}
            call_wall = md_data.get("call_wall")
            put_wall  = md_data.get("put_wall")
            flip      = md_data.get("flip")
            bias      = ""

        # 3. 計算盈虧
        if price:
            holding_val  = price * qty
            holding_cost = cost  * qty
            pnl          = holding_val - holding_cost
            pnl_pct      = pnl / holding_cost * 100 if holding_cost else 0
            total_pnl   += pnl
            total_value += holding_val
            total_cost  += holding_cost
        else:
            pnl = pnl_pct = 0

        # 4. GEX狀態圖標
        if price and call_wall and put_wall:
            dist_call = (call_wall - price) / price * 100 if price > 0 else 99
            dist_put  = (price - put_wall)  / price * 100 if price > 0 else 99
            if dist_call < 3:
                gex_icon = "⚠️"
            elif dist_put < 3:
                gex_icon = "🛡️"
            elif bias == "SHORT_GAMMA":
                gex_icon = "📉"
            else:
                gex_icon = "▫️"
        else:
            gex_icon = "❓"

        pnl_color = "🟢" if pnl >= 0 else "🔴"
        pnl_arrow = "▲" if pnl >= 0 else "▼"
        price_str = f"${price:,.2f}" if price else "N/A"
        pnl_str   = f"{pnl_color}{pnl_arrow}${abs(pnl):,.0f}({pnl_pct:+.1f}%)" if pnl != 0 else "—"

        # GEX牆距離小摘要
        wall_str = ""
        if price and call_wall and put_wall:
            wall_str = f"  _P:{put_wall:,.0f}↔C:{call_wall:,.0f}_"

        lines.append(f"{gex_icon} *{sym}* {price_str}  {pnl_str}{wall_str}")

    total_color = "🟢" if total_pnl >= 0 else "🔴"
    total_arrow = "▲" if total_pnl >= 0 else "▼"
    total_ret   = (total_pnl / total_cost * 100) if total_cost else 0
    lines.append(f"\n💰 *總浮盈虧：{total_color}{total_arrow}${abs(total_pnl):,.0f}（{total_ret:+.1f}%）*")

    await update.message.reply_text("\n".join(lines), parse_mode="Markdown")

async def cmd_alert(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """
    用法：
      /alert NVDA above 210   — NVDA升穿210提示
      /alert NVDA below 190   — NVDA跌穿190提示
      /alert list             — 列出所有警報
      /alert remove NVDA      — 移除NVDA警報
    """
    register_chat(update.effective_chat.id)
    if not context.args:
        await update.message.reply_text(
            "用法：\n"
            "/alert NVDA above 210\n"
            "/alert NVDA below 190\n"
            "/alert list\n"
            "/alert remove NVDA"
        )
        return

    alerts = load_alerts()
    chat_id = str(update.effective_chat.id)
    user_alerts = alerts.get(chat_id, {})

    if context.args[0].lower() == "list":
        if not user_alerts:
            await update.message.reply_text("📭 暫無設定警報")
            return
        lines = ["🔔 *目前警報*\n"]
        for sym, conf in user_alerts.items():
            lines.append(f"{sym}: {conf['direction']} ${conf['price']}")
        await update.message.reply_text("\n".join(lines), parse_mode="Markdown")
        return

    if context.args[0].lower() == "remove":
        if len(context.args) < 2:
            await update.message.reply_text("用法：/alert remove NVDA")
            return
        sym = context.args[1].upper()
        if sym in user_alerts:
            del user_alerts[sym]
            alerts[chat_id] = user_alerts
            save_alerts(alerts)
            await update.message.reply_text(f"✅ 已移除 {sym} 警報")
        else:
            await update.message.reply_text(f"⚠️ {sym} 沒有設定警報")
        return

    if len(context.args) < 3:
        await update.message.reply_text("用法：/alert NVDA above 210")
        return

    sym       = context.args[0].upper()
    direction = context.args[1].lower()
    try:
        price = float(context.args[2])
    except ValueError:
        await update.message.reply_text("⚠️ 價格必須是數字")
        return

    if sym not in VALID_SYMS:
        await update.message.reply_text(f"⚠️ {sym} 不在持倉清單")
        return
    if direction not in ("above", "below"):
        await update.message.reply_text("⚠️ 方向必須是 above 或 below")
        return

    user_alerts[sym] = {"direction": direction, "price": price}
    alerts[chat_id]  = user_alerts
    save_alerts(alerts)

    await update.message.reply_text(
        f"✅ 已設定：{sym} {direction} ${price} 時提示"
    )

    return {k: v for k, v in log.items() if v >= cutoff}

def check_gex_proximity(sym: str, price: float, call_wall: float, put_wall: float) -> dict | None:

    計算現價相對Call/Put Wall嘅位置，兩種情況都會觸發：
      1.「逼近」— 現價仲喺wall嗰邊未過界，但距離 < GEX_ALERT_THRESHOLD_PCT（原有邏輯）
      2.「已跌穿/已突破」— 現價已經過咗個wall（v1.5新增）

    v1.5【重要bugfix】：2026-07-23實測GOOGL績後大跌完全冇觸發任何警報——
    舊版淨係接受 `0 <= dist_pct < threshold`，即係話現價一定要仲企喺wall
    嗰邊、由外面逼近先會觸發；現價一旦已經跌穿Put Wall／突破Call Wall
    （dist_pct變負數），呢個判斷就永遠唔會再成立，個wall等同「隱形」。
    根查發現觸發GOOGL事件嘅根源有兩層疊埋：(a) 全部18隻嘅.md每日快照
    pipeline已停咗成個月冇更新，(b) GOOGL績後由.md嗰陣嘅價位一路跌落嚟，
    已經一次過插穿咗stale .md記錄嗰條Put Wall——用返嗰份舊wall數據計，
    現價已經低過個wall，`0 <= dist_put_pct`呢個條件唔成立，觸發判斷直接
    靜音。同一時間AVGO/VRT淨係因為跌幅細啲、僥倖仲企喺各自stale wall
    之上先觸發到，証明呢個唔係GOOGL獨有、而係任何一隻股票跌穿/升穿wall
    之後都會撞到嘅通用缺陷。
    而家已跌穿/已突破一律照樣觸發（wall_type會帶埋"_breach"標記俾
    gex_alert_job()分辨），唔會再因為「已經太遲」而完全靜音；同時保留
    原有「逼近」觸發，兩者可以並存。

    若同時有多個觸發（理論上罕見），優先順序：已突破 > 逼近，
    再各自揀距離/突破幅度最細嗰個（最貼近現價、最新鮮嗰單先講）。
    """
    if not price or not call_wall or not put_wall:
        return None

    # v1.25 sanity guard：Put Wall高過Call Wall，或任何wall偏離現價>40%，視為GEX數據失真
    if put_wall > call_wall:
        return None
    if (abs(call_wall - price) / price * 100 > GEX_MAX_WALL_DEVIATION_PCT or
            abs(price - put_wall) / price * 100 > GEX_MAX_WALL_DEVIATION_PCT):
        return None

    dist_call_pct = (call_wall - price) / price * 100   # 現價喺call_wall之下 → 正數
    dist_put_pct  = (price - put_wall)  / price * 100   # 現價喺put_wall之上 → 正數

    triggers = []  # (wall_type, dist_pct, wall_price, is_breach)

    # ── 逼近（原有邏輯：未過界，喺threshold%範圍內）──
    if 0 <= dist_call_pct < GEX_ALERT_THRESHOLD_PCT:
        triggers.append(("call", dist_call_pct, call_wall, False))
    if 0 <= dist_put_pct < GEX_ALERT_THRESHOLD_PCT:
        triggers.append(("put", dist_put_pct, put_wall, False))

    # ── 已突破/已跌穿（v1.5新增，唔設幅度上限——過咗界本身就係重點）──
    if dist_call_pct < 0:   # 現價 > call_wall，已突破阻力
        triggers.append(("call", abs(dist_call_pct), call_wall, True))
    if dist_put_pct < 0:    # 現價 < put_wall，已跌穿支撐
        triggers.append(("put", abs(dist_put_pct), put_wall, True))

    if not triggers:
        return None

    # 優先揀「已突破」，再揀距離/幅度最細嗰個
    triggers.sort(key=lambda t: (not t[3], t[1]))
    wall_type, dist_pct, wall_price, is_breach = triggers[0]
    return {
        "wall_type": wall_type, "dist_pct": dist_pct,
        "wall_price": wall_price, "is_breach": is_breach,
    }

# v1.9新增：GEX預熱job嘅並行度上限。背景見下面gex_refresh_job()
# docstring——18隻齊齊thread撞落futu_bridge_v2.py單一
# OpenQuoteContext connection（_ctx全域變量），會令部分request
# 「快速失敗」而唔係真係timeout，而家降低到4，等request分批打，
# 減少同一時間撞_ctx嘅機會。
GEX_REFRESH_MAX_WORKERS = 4


def _run_gex_refresh_sync() -> int:
    """
    v1.16新增：抽出咗gex_refresh_job()原本喺coroutine入面直接做嘅
    阻塞邏輯（ThreadPoolExecutor + concurrent.futures.as_completed()，
    最長等180秒）。純同步function，冇任何async/await，設計俾
    asyncio.to_thread()放入獨立線程執行——詳細背景見
    gex_refresh_job() docstring嘅v1.16段落。
    """
    def _fetch_one_gex(sym: str) -> tuple[str, dict | None]:
        """thread worker：攞單隻GEX，45秒timeout"""
        try:
            r = requests.get(
                f"{FUTU_BRIDGE_URL}/api/gex/{sym.upper()}",
                timeout=45
            )
            if r.status_code == 200:
                d = r.json()
                if d.get("call_wall") and d.get("put_wall"):
                    return sym, d
                # v1.13新增：200 OK但call_wall/put_wall缺失——舊版靜默
                # 略過，而家印返bridge嘅error欄位（如有），方便排查
                print(f"   ⚠️ {sym} GEX 200 OK但call_wall/put_wall缺失"
                      f"（bridge error: {d.get('error', '無error欄位')}）")
        except Exception as e:
            print(f"   ⚠️ {sym} GEX timeout/error: {type(e).__name__}")
        return sym, None

    success = 0
    # v1.9：max_workers由len(ALL_HOLDINGS)（18，全部齊齊撞）降低到
    # GEX_REFRESH_MAX_WORKERS（4，分批），見gex_refresh_job() docstring
    # 根源分析
    with concurrent.futures.ThreadPoolExecutor(max_workers=GEX_REFRESH_MAX_WORKERS) as pool:
        futures = {pool.submit(_fetch_one_gex, h["sym"]): h["sym"] for h in ALL_HOLDINGS}
        # v1.9：整體timeout由50秒放寬到180秒——降低並行度後18隻全部
        # 行完嘅總時間會拉長（分批排隊），但job本身30分鐘先跑一次，
        # 呢個等待時間完全可以接受
        for future in concurrent.futures.as_completed(futures, timeout=180):
            try:
                sym, data = future.result()
                if data:
                    set_gex_live(sym, data)
                    success += 1
                    print(f"   ✅ {sym}: CW=${data['call_wall']:,.0f} PW=${data['put_wall']:,.0f}")
            except Exception as e:
                print(f"   ⚠️ future error: {e}")
    return success


async def gex_refresh_job(context: ContextTypes.DEFAULT_TYPE):
    """
    每30分鐘預熱 GEX cache（分批並行攞18隻，唔 block alert job）

    │ 幾十秒（唔再係6秒），但job本身30分鐘先跑一次，  │
    │ 完全負擔得起                                     │
    │ 成功 → 存入 _gex_live_cache（20小時TTL，v1.13起  │
    │        同步持久化落磁碟）                        │
    │ 失敗 → 保留舊cache / alert job fallback .md     │
    └─────────────────────────────────────────────────┘

    v1.9【重要修正】：2026-08-05從bot terminal log發現一個穩定
    Pattern——18隻齊齊fire request嗰陣，成個job淨係使6秒左右就
    完成（14:37:52→14:37:58／15:37:52→15:37:58），但成功率穩定

    14:37嗰輪失敗，15:37嗰輪又成功；PLTR兩輪都成功）。呢個組合
    （完成極快 + 失敗率穩定 + 失敗symbol隨機轉換）唔似係真係45秒
    request timeout（如果係真timeout，18個thread嘅batch理應接近
    45秒先完成）——更似係futu_bridge_v2.py嗰邊單一
    OpenQuoteContext（`_ctx`全域變量，見get_ctx()）被18條thread
    同時撞：FastAPI/uvicorn雖然可以同時處理多個incoming

# 期權到期日曆提醒（v1.20新增）── 每日一次，08:00 HKT
# ═══════════════════════════════════════════════════
OPEX_REMINDER_DAYS_AHEAD = 3   # 提前幾多日開始提醒
_OPEX_TYPE_ICON = {"monthly": "🟢", "quarterly": "🟣"}
_OPEX_TYPE_LABEL = {"monthly": "Monthly OPEX", "quarterly": "Quarterly（Triple Witching）"}

async def opex_reminder_job(context: ContextTypes.DEFAULT_TYPE):
    """
    每日一次（08:00 HKT，見main()嘅run_daily註冊）背景job：check全部
    持倉未來OPEX_REMINDER_DAYS_AHEAD日內有冇Monthly/Quarterly OPEX，
    有就主動推送提醒。

    呼應futu_bridge_v2.py v3.5新增嘅/api/options_calendar——
    gex_chart_v5_terminal.html v5.17嘅「📅 到期日曆」tab用緊同一個
    endpoint，Bot呢度直接重用（見fetch_options_calendar()），兩個
    surface（網頁被動查詢 + Telegram主動推送）共享同一份數據源同
    Weekly/Monthly/Quarterly分類邏輯，唔使Bot呢邊獨立開發一套。

    背景：GEX Wall建基於期權OI，OPEX一過，相關到期日嘅OI會「roll
    off」，Call/Put Wall好可能大幅跳動，值得預先留意。Weekly到期
    太頻密（幾乎每個星期五都有），唔值得日日提醒——同gex_chart_v5_
    terminal.html「到期日曆」摘要橫額用緊嘅篩選邏輯一致，淨係篩選
    type為monthly/quarterly嘅事件。

    冷卻：同一個「sym_date」組合只會提醒一次（用opex_reminder_log.json
    記錄，同gex_alert_log.json同一套JSON persist模式），避免OPEX_
    REMINDER_DAYS_AHEAD日窗口內每日重跑都重複推送同一單。
    """
    known_chats = get_known_chats()
    if not known_chats:
        print(f"[{datetime.datetime.now().strftime('%H:%M:%S')}] 📅 OPEX提醒：暫無已知chat_id，略過")
        return

    events = fetch_options_calendar(days_ahead=OPEX_REMINDER_DAYS_AHEAD)
    if events is None:
        print(f"[{datetime.datetime.now().strftime('%H:%M:%S')}] 📅 OPEX提醒：Bridge離線或連接失敗，略過")
        return

    soon = [e for e in events if e.get("type") in ("monthly", "quarterly")]
    if not soon:
        print(f"[{datetime.datetime.now().strftime('%H:%M:%S')}] 📅 OPEX提醒：未來{OPEX_REMINDER_DAYS_AHEAD}日內冇Monthly/Quarterly OPEX")
        return

    log = load_opex_reminder_log()
    new_events = [e for e in soon if f"{e['sym']}_{e['date']}" not in log]
    if not new_events:
        print(f"[{datetime.datetime.now().strftime('%H:%M:%S')}] 📅 OPEX提醒：{len(soon)}個事件已提醒過，冷卻中")
        return

    lines = ["📅 *期權到期日曆提醒*\n"]
    for e in sorted(new_events, key=lambda x: x["date"]):
        icon     = _OPEX_TYPE_ICON.get(e["type"], "⚪")
        type_lbl = _OPEX_TYPE_LABEL.get(e["type"], e["type"])
        due      = "今日" if e["days_to_expiry"] == 0 else f"{e['days_to_expiry']}日後"
        lines.append(f"{icon} *{e['sym']}* — {e['date']}（{due}）　{type_lbl}")
    lines.append(
        "\n_OPEX過後相關到期日OI會roll off，Call/Put Wall可能大幅跳動，"
        "查詢 /gex 時留意Wall有冇突然改變_"
    )
    msg = "\n".join(lines)

    for chat_id in known_chats:
        try:
            await context.bot.send_message(chat_id=int(chat_id), text=msg, parse_mode="Markdown")
        except Exception as e:
            print(f"   ⚠️ OPEX提醒推送chat_id={chat_id}失敗: {e}")

    now_ts = time.time()
    for e in new_events:
        log[f"{e['sym']}_{e['date']}"] = now_ts
    save_opex_reminder_log(log)

    print(f"[{datetime.datetime.now().strftime('%H:%M:%S')}] 📅 OPEX提醒已推送 {len(new_events)} 個事件（{', '.join(e['sym'] for e in new_events)}）")

_SESS_SHORT = {"premarket": "盤前", "regular": "盤中", "afterhours": "盤後",
               "overnight": "夜盤", "weekend": "週末"}

async def gex_alert_job(context: ContextTypes.DEFAULT_TYPE):
    """
    每15分鐘背景job（快速版）：
    ┌──────────────────────────────────────────────┐
    │  現價  → Futu /api/quote 批量1個call（快）  │
    │  GEX牆 → .md 內存緩存（60分鐘TTL，即時）   │
    │  完全唔call fetch_futu_gex()（太慢！       │
    │  每隻30秒×18隻=9分鐘，必定超時）            │
    └──────────────────────────────────────────────┘
    GEX牆每日更新一次完全夠用——警報精度在於現價
    是否真的接近牆，牆本身唔係每分鐘都移動。
    Futu離線時自動fallback Finnhub，唔會靜默失效。

    v1.1：fetch_all_quotes_futu() 而家已經係session-aware（見
    fetch_all_quotes_futu docstring），警報訊息會顯示緊用緊邊個
    session嘅價（例如「Futu·盤後」），方便肉眼核實盤前/盤後/夜盤
    警報係咪真係用緊即時價、唔再係得個name叫「Futu」但實際上係
    舊last_price。
    """
    # ── 休市日跳過（週末 / NYSE 假期）──
    if not is_nyse_trading_day():
        et_now = datetime.datetime.now(US_TZ)
        print(f"[{et_now.strftime('%H:%M')} ET] 🗓 NYSE休市（{et_now.strftime('%a %Y-%m-%d')}），跳過GEX警報")
        return

    ts_str = datetime.datetime.now().strftime('%H:%M:%S')
    print(f"[{ts_str}] 🔍 GEX警報檢查...")

    known_chats = get_known_chats()
    if not known_chats:
        print("   ⚠️ 暫無已知chat_id，略過")
        return

    # ── Step 1：批量攞現價（Futu T1，1個call）──
    quotes  = fetch_all_quotes_futu()
    futu_ok = bool(quotes)
    if not futu_ok:
        print("   ℹ️ Futu Bridge 離線，改用 Finnhub 報價")

    alert_log       = load_gex_alert_log()
    now_ts          = datetime.datetime.now().timestamp()
    triggered_count = 0
    skipped_no_wall = 0
    skipped_no_price = 0

    for h in ALL_HOLDINGS:
        sym  = h["sym"]
        name = h["name"]

        # ── Step 2：現價（Futu批量 → Finnhub逐隻fallback）──
        if futu_ok and sym in quotes:
            q     = quotes[sym]
            price = q["price"]
            pct   = q.get("pct", 0)
            sess  = q.get("session", "")
            psrc  = f"Futu·{_SESS_SHORT.get(sess, sess)}" if sess else "Futu"
        else:
            live = fetch_live_quote(sym)          # Futu單隻 → Finnhub
            if not live or not live.get("price"):
                skipped_no_price += 1
                continue
            price = live["price"]
            pct   = live.get("pct", 0)
            src   = live.get("source", "?")
            sess  = live.get("session", "")
            psrc  = f"Futu·{_SESS_SHORT.get(sess, sess)}" if src == "futu_t1" and sess else src

        # ── Step 3：GEX牆（優先 Futu live cache → fallback .md）──
        # _gex_live_cache 由 gex_refresh_job 每30分鐘更新（即時Futu數據）
        # .md 係每日Barchart快照，作保底fallback
        gex_live = get_gex_live(sym)
        if gex_live and gex_live.get("call_wall") and gex_live.get("put_wall"):
            call_wall = gex_live["call_wall"]
            put_wall  = gex_live["put_wall"]
            flip      = gex_live.get("flip")
            wall_src  = "Futu即時"
        else:
            md = fetch_stock_md(sym)              # 60分鐘內存緩存，極快
            if not md:
                skipped_no_wall += 1
                continue
            md_data   = parse_md(md)
            call_wall = md_data.get("call_wall")
            put_wall  = md_data.get("put_wall")
            flip      = md_data.get("flip")
            wall_src  = ".md快照"

        if not call_wall or not put_wall:
            skipped_no_wall += 1
            print(f"   ⚠️ {sym}: 無GEX牆數據（wall_src={wall_src}）")
            continue

        # ── Step 4：proximity 檢查 ──
        result = check_gex_proximity(sym, price, call_wall, put_wall)
        if not result:
            continue

        wall_type  = result["wall_type"]
        dist_pct   = result["dist_pct"]
        wall_price = result["wall_price"]
        is_breach  = result["is_breach"]   # v1.5新增

        # ── Step 5：冷卻（1小時內不重複）──
        # v1.5：log_key加返"_breach"後綴，令「逼近」轉「已突破」呢個狀態
        # 轉變本身當做新事件、唔會俾返舊嘅「逼近」冷卻期擋住唔發——
        # 呢個轉變（例如GOOGL由「仲喺Put Wall之上1.55%」變成「已經跌穿」）
        # 正正係最需要即時通知嘅時刻。
        log_key = f"{sym}_{wall_type}" + ("_breach" if is_breach else "")
        if now_ts - alert_log.get(log_key, 0) < GEX_ALERT_COOLDOWN_SEC:
            continue

        # ── Step 6：推送（v1.3：Call Wall／Put Wall／Gamma Flip 三個水平
        #    同時顯示，唔再淨係顯示觸發嗰邊嗰隻——call_wall/put_wall/flip
        #    喺Step 3已經攞齊，呢度直接攞嚟用；觸發嗰個用⚠️標示。
        #    v1.5：已突破/已跌穿用更搶眼嘅icon + 文字區分,唔再同「逼近」
        #    共用「距離...僅X%」呢句唔啱意思嘅描述）──
        wall_label = "Call Wall（阻力）" if wall_type == "call" else "Put Wall（支撐）"
        arrow      = "▲" if pct >= 0 else "▼"
        call_mark  = " ⚠️觸發" if wall_type == "call" else ""
        put_mark   = " ⚠️觸發" if wall_type == "put"  else ""
        flip_str   = f"\nGamma Flip：${flip:,.2f}" if flip else ""
        src_note   = f"_{wall_src} GEX牆_"

        if is_breach:
            icon        = "🚨" if wall_type == "call" else "💥"
            verb        = "已突破" if wall_type == "call" else "已跌穿"
            action_line = f"{verb}{wall_label}：{dist_pct:.1f}%（超出wall本身嘅幅度）"
        else:
            icon        = "🔴" if wall_type == "call" else "🟢"
            action_line = f"距離{wall_label}：僅 {dist_pct:.1f}%"

        msg = (
            f"{icon} *GEX警報：{sym}* — {name}\n\n"
            f"現價（{psrc}）：${price:,.2f}　{arrow}{abs(pct):.2f}%\n"
            f"{action_line}\n\n"
            f"Call Wall（阻力）：${call_wall:,.2f}{call_mark}\n"
            f"Put Wall（支撐）：${put_wall:,.2f}{put_mark}"
            f"{flip_str}\n"
            f"{src_note}\n\n"
            f"_此警報1小時內不會重複發送_"
        )

        for chat_id in known_chats:
            try:
                await context.bot.send_message(
                    chat_id=int(chat_id), text=msg, parse_mode="Markdown"
                )
            except Exception as e:
                print(f"   ⚠️ 推送chat_id={chat_id}失敗: {e}")

        alert_log[log_key] = now_ts
        triggered_count   += 1
        breach_tag = "（已突破）" if is_breach else "（逼近）"
        print(f"   🔔 觸發{breach_tag}：{sym} {wall_label} {dist_pct:.1f}% [{psrc}]")

    if triggered_count > 0:
        save_gex_alert_log(alert_log)

    print(
        f"   ✅ 完成 | 警報:{triggered_count} | "
        f"無牆:{skipped_no_wall} | 無價:{skipped_no_price} | "
        f"報價來源:{'Futu T1' if futu_ok else 'Finnhub'}"
    )

# /ask 指令：自由問任何投資問題，AI結合持倉上下文回答
# ═══════════════════════════════════════════════════
# 識別問題中提到嘅股票代號（持倉 + 常見watchlist）
_WATCHLIST_EXTRA = {"SPCX", "TSLA", "AMD", "INTC", "IONQ", "RGTI", "TXN"}

def _extract_syms_from_text(text: str) -> list[str]:
    """從自然語言問題中提取可能嘅股票代號（2-5個英文大寫字母）"""
    known = {h["sym"] for h in ALL_HOLDINGS} | _WATCHLIST_EXTRA
    words = text.upper().replace("？","").replace("?","").replace("，","").replace(",","").split()
    found = []
    for w in words:
        clean = w.strip(".,!！。、")
        if clean in known:
            found.append(clean)
        elif 2 <= len(clean) <= 5 and clean.isalpha() and clean not in {"THE","AND","FOR","ARE","BUT","NOT","YOU","ALL","CAN","HAS","WAS","HOW"}:
            found.append(clean)
    # 去重,保持順序
    seen = set()
    result = []
    for s in found:
        if s not in seen:
            seen.add(s)
            result.append(s)
    return result[:3]  # 最多3隻


def _fetch_news_summary(sym: str) -> str:
    """攞Finnhub最新新聞標題(最多3條)做AI上下文用"""
    try:
        import datetime as dt
        to_date   = dt.date.today().isoformat()
        from_date = (dt.date.today() - dt.timedelta(days=7)).isoformat()
        url = f"https://finnhub.io/api/v1/company-news?symbol={sym}&from={from_date}&to={to_date}&token={FINNHUB_KEY}"
        r = requests.get(url, timeout=8)
        if r.status_code == 200:
            items = r.json()[:3]
            if items:
                headlines = [f"- {n.get('headline','')}" for n in items if n.get('headline')]
                return "\n".join(headlines)
    except Exception:
        pass
    return ""

    fetch_futu_gex()攞Futu即時GEX（同/gex一致），.md做fallback，

用戶問題：{question}

請用繁體中文回答（可加少量粵語表達），語氣專業、具體、直接。
回覆控制在200字以內，唔好用免責聲明套話。

# /start 指令
# ═══════════════════════════════════════════════════
async def cmd_start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    register_chat(update.effective_chat.id)
    await update.message.reply_text(
        "👋 歡迎使用GEX持倉監控Bot！\n輸入 /help 查看所有指令\n\n"
        "🔔 自動GEX警報已為你啟用，當持倉逼近Call/Put Wall時會主動通知"
    )

# ═══════════════════════════════════════════════════
# 未知指令（避免Telegram靜默無回應）
# ═══════════════════════════════════════════════════
async def cmd_unknown(update: Update, context: ContextTypes.DEFAULT_TYPE):
    register_chat(update.effective_chat.id)
    text = update.message.text or ""
    await update.message.reply_text(
        f"⚠️ 未知指令：{text}\n"
        "輸入 /help 查看可用指令清單"
    )

# ═══════════════════════════════════════════════════
# 主程式
# ═══════════════════════════════════════════════════
def main():
    print(f"🤖 GEX Telegram Bot v{BOT_VERSION} 啟動中...")

    # v1.13新增：啟動時讀返上次留低嘅GEX live cache（JSON持久化），
    # 令process重啟唔會令20小時TTL形同虛設（見上面_gex_live_cache
    # 區塊嘅v1.13註解）
    _load_gex_live_cache_from_disk()

    # 自我檢查：確保key已填好，唔再係佔位字
    if TELEGRAM_BOT_TOKEN == "YOUR_TELEGRAM_BOT_TOKEN" or not TELEGRAM_BOT_TOKEN:
        print("❌ 錯誤：TELEGRAM_BOT_TOKEN 仍未填入，請編輯腳本第44行")
        return
    if OPENROUTER_API_KEY == "YOUR_OPENROUTER_API_KEY" or not OPENROUTER_API_KEY:
        print("⚠️ 警告：OPENROUTER_API_KEY 仍未填入，AI分析功能將無法使用")
        print("   （/gex /report /pnl 等基本功能不受影響，只係AI分析部分會錯誤）")

    # 延長逾時時間（預設5秒對某些Windows網絡環境太短）
    # 並指明用IPv4，避免IPv6解析延遲導致逾時
    request = HTTPXRequest(
        connect_timeout=30.0,
        read_timeout=30.0,
        write_timeout=30.0,
        pool_timeout=30.0,
    )

    app = (
        Application.builder()
        .token(TELEGRAM_BOT_TOKEN)
        .request(request)
        .get_updates_request(request)
        .build()
    )

    app.add_handler(CommandHandler("start",  cmd_start))
    app.add_handler(CommandHandler("help",   cmd_help))
    app.add_handler(CommandHandler("gex",    cmd_gex))
    app.add_handler(CommandHandler("cot",    cmd_cot))
    app.add_handler(CommandHandler("report", cmd_report))
    app.add_handler(CommandHandler("pnl",    cmd_pnl))
    app.add_handler(CommandHandler("alert",  cmd_alert))
    app.add_handler(CommandHandler("ask",    cmd_ask))

    # GEX 預熱 job（每30分鐘，並行攞18隻Futu即時GEX）
    app.job_queue.run_repeating(
        gex_refresh_job,
        interval=1800,   # 30 分鐘
        first=60,        # 啟動後60秒先跑第一次（等OpenD穩定）
        name="gex_refresh"
    )
    # 捕捉所有未匹配嘅指令（必須放最後，唔阻擋上面已註冊嘅指令）
    app.add_handler(MessageHandler(filters.COMMAND, cmd_unknown))

    # 背景GEX警報：每15分鐘檢查一次，啟動後10秒先首次執行
    if app.job_queue:
        app.job_queue.run_repeating(
            gex_alert_job,
            interval=900,    # 15分鐘 = 900秒
            first=10,         # 啟動後10秒首次執行
        )
        print("🔔 自動GEX警報已啟用（每15分鐘檢查，距離Wall<3%觸發）")

        # v1.20新增：期權到期日曆提醒，每日08:00 HKT執行一次。用
        # Asia/Hong_Kong做tzinfo（HKT全年UTC+8冇夏令時，唔似ET咁
        # 要理會DST），時間揀朝早，等用戶開市前有機會睇到提醒、
        # 及早decide要唔要喺OPEX前調整持倉。
        app.job_queue.run_daily(
            opex_reminder_job,
            time=datetime.time(hour=8, minute=0, tzinfo=ZoneInfo("Asia/Hong_Kong")),
            name="opex_reminder",
        )
        print("📅 期權到期日曆提醒已啟用（每日08:00 HKT，Monthly/Quarterly OPEX提前3日提醒）")

        # v1.21新增：CFTC COT週報自動推送，每週六08:15 HKT執行一次
        # （days=(5,)——PTB JobQueue.run_daily嘅days參數用Python
        # weekday慣例：0=週一...6=週日，5=週六）。刻意排喺Windows
        # 工作排程器COT_Monitor_Weekly（週六08:00 HKT）之後15分鐘，
        # 留返buffer等cftc_cot_updater.py行完、GitHub cot/*.md
        # 已經更新先至嚟fetch。
        app.job_queue.run_daily(
            cot_reminder_job,
            time=datetime.time(hour=8, minute=15, tzinfo=ZoneInfo("Asia/Hong_Kong")),
            days=(5,),
            name="cot_reminder",
        )
        print("📊 CFTC COT週報提醒已啟用（每週六08:15 HKT，Leveraged Funds淨倉）")
    else:
        print("⚠️ job_queue未能初始化，自動GEX警報功能停用")
        print("   請確認已安裝：pip install \"python-telegram-bot[job-queue]\"")

    print("✅ Bot 已啟動，等待Telegram指令...")
    print("   （按 Ctrl+C 停止）")
    app.run_polling()

if __name__ == "__main__":
    main()
